-- True Ingenium-only base-table count
SELECT COUNT(*) AS ing_curr_base_tables
FROM PROD_DV.INFORMATION_SCHEMA.TABLES
WHERE TABLE_SCHEMA='BUSINESS_VAULT'
  AND TABLE_NAME LIKE '%_CURR'
  AND TABLE_NAME LIKE 'P\_ING\_%' ESCAPE '\'
  AND TABLE_TYPE='BASE TABLE';

-- Is there ANY view-type _CURR at all, anywhere?
SELECT TABLE_SCHEMA, TABLE_TYPE, COUNT(*)
FROM PROD_DV.INFORMATION_SCHEMA.TABLES
WHERE TABLE_NAME LIKE '%_CURR'
GROUP BY TABLE_SCHEMA, TABLE_TYPE;
