SELECT COALESCE(NULLIF(TRIM(S.ELIG_CD),''),'<blank>') AS SNAPSHOT_ELIG,
       COALESCE(NULLIF(TRIM(C.ELIG_CD),''),'<blank>') AS CURRENT_ELIG,
       COUNT(*) AS POLICIES
FROM   KCDWHPRC.T8LT1_AV111_0930_V2 S
JOIN   KCDWHPRC.T8LT1 C
       ON  C.CO_ID = S.CO_ID AND C.POL_ID = S.POL_ID
       AND C.LNG_TRM_CARE_POL_ID = S.LNG_TRM_CARE_POL_ID
       AND C.CRNT_REC_IND = 'Y'
WHERE  COALESCE(TRIM(S.ELIG_CD),'') <> COALESCE(TRIM(C.ELIG_CD),'')
GROUP BY COALESCE(NULLIF(TRIM(S.ELIG_CD),''),'<blank>'),
         COALESCE(NULLIF(TRIM(C.ELIG_CD),''),'<blank>')
ORDER BY POLICIES DESC;
